import fs from 'node:fs';
import { countTokens, type EncodingName } from '@/lib/omega/bpe';
import { aetherionEncode, aetherionDecode, aetherionDecoderPrompt } from '@/lib/omega/aetherion';
import { CHAOS_900, CHAOS_G_CJK, CHAOS_F_LLM_REPORT, MOSAIC_HANDTRACE_300, mosaicFixtures } from './fixtures';

async function runRedTeam() {
  console.log('=== AETHERION-Ω RED-TEAM VERIFICATION SUITE ===\n');
  const enc: EncodingName = 'o200k_base';
  const f = mosaicFixtures();

  const holdouts = fs.readdirSync('bench/holdout').sort().map(nm => ({
    name: 'holdout/' + nm,
    text: fs.readFileSync('bench/holdout/' + nm, 'utf8'),
  }));

  const fixtureCases = [
    { name: 'CHAOS_900', text: CHAOS_900 },
    { name: 'CHAOS_G_CJK', text: CHAOS_G_CJK },
    { name: 'CHAOS_F_LLM_REPORT', text: CHAOS_F_LLM_REPORT },
    { name: 'MOSAIC_HANDTRACE_300', text: MOSAIC_HANDTRACE_300 },
    { name: 'json-log-40', text: f.jsonLog },
    { name: 'csv-60', text: f.csv },
    { name: 'chat-48', text: f.chat },
    { name: 'grid-30', text: f.grid },
    { name: 'rle-1400', text: f.rle },
    { name: 'idrun-200', text: f.idrun },
    { name: 'prose', text: f.prose },
  ];

  let passed = 0;
  let failed = 0;

  console.log('--- GATE 1: HOLDOUT SUITE EXACT RESTORATION & TOKEN REDUCTION ---');
  let holdoutInToks = 0;
  let holdoutOutToks = 0;

  for (const c of holdouts) {
    const inT = countTokens(c.text, enc);
    holdoutInToks += inT;
    const res = aetherionEncode(c.text, enc);
    holdoutOutToks += res.outTokens;

    const roundTrip = aetherionDecode(res.wire);
    const exact = roundTrip === c.text;

    if (exact) {
      passed++;
      console.log(`[PASS] ${c.name.padEnd(30)} in:${inT} -> out:${res.outTokens} (saved ${inT - res.outTokens} tok, ${res.savingsPct}%) [mode: ${res.mode}]`);
    } else {
      failed++;
      console.error(`[FAIL] ${c.name.padEnd(30)} decoded mismatch!`);
    }
  }

  const netHoldoutSaved = holdoutInToks - holdoutOutToks;
  const netHoldoutPct = Math.round((netHoldoutSaved / holdoutInToks) * 1000) / 10;
  console.log(`\nHOLDOUT TOTALS: ${holdoutInToks} raw -> ${holdoutOutToks} AETHERION-Ω (Saved ${netHoldoutSaved} tok, ${netHoldoutPct}% net reduction)\n`);

  console.log('--- GATE 2: FIXTURE SUITE EXACT RESTORATION & HIGH-COMPRESSION LANES ---');
  let fixtureInToks = 0;
  let fixtureOutToks = 0;

  for (const c of fixtureCases) {
    const inT = countTokens(c.text, enc);
    fixtureInToks += inT;
    const res = aetherionEncode(c.text, enc);
    fixtureOutToks += res.outTokens;

    const roundTrip = aetherionDecode(res.wire);
    const exact = roundTrip === c.text;

    if (exact) {
      passed++;
      console.log(`[PASS] ${c.name.padEnd(30)} in:${inT} -> out:${res.outTokens} (saved ${inT - res.outTokens} tok, ${res.savingsPct}%) [mode: ${res.mode}]`);
    } else {
      failed++;
      console.error(`[FAIL] ${c.name.padEnd(30)} decoded mismatch!`);
    }
  }

  const netFixtureSaved = fixtureInToks - fixtureOutToks;
  const netFixturePct = Math.round((netFixtureSaved / fixtureInToks) * 1000) / 10;
  console.log(`\nFIXTURE TOTALS: ${fixtureInToks} raw -> ${fixtureOutToks} AETHERION-Ω (Saved ${netFixtureSaved} tok, ${netFixturePct}% net reduction)\n`);

  console.log('--- GATE 3: MODEL-READABILITY & SYSTEM PROMPT CONTRACT ---');
  const sampleWire = aetherionEncode(f.prose + '\n' + f.prose, enc).wire;
  const prompt = aetherionDecoderPrompt(sampleWire);
  if (prompt.includes('AETHERION-Ω PROTOCOL') && sampleWire.length > 0) {
    passed++;
    console.log('[PASS] Model-readability prompt verified.');
  } else {
    failed++;
    console.error('[FAIL] Prompt verification failed.');
  }

  console.log(`\n==================================================`);
  console.log(`RED-TEAM SUMMARY: ${passed} PASSED, ${failed} FAILED`);
  console.log(`==================================================`);

  if (failed > 0) process.exit(1);
}

runRedTeam().catch(err => {
  console.error(err);
  process.exit(1);
});

async function main() {
  const enc = (process.argv[2] as EncodingName) || 'o200k_base';
  const f = mosaicFixtures();
  const lanes: Array<[string, string]> = [
    ...fs.readdirSync('bench/holdout').sort().map(nm => ['holdout/' + nm, fs.readFileSync('bench/holdout/' + nm, 'utf8')] as [string, string]),
    ['CHAOS_900', CHAOS_900],
    ['CHAOS_G_CJK', CHAOS_G_CJK],
    ['CHAOS_F_LLM_REPORT', CHAOS_F_LLM_REPORT],
    ['MOSAIC_HANDTRACE_300', MOSAIC_HANDTRACE_300],
    ['json-log-40', f.jsonLog],
    ['csv-60', f.csv],
    ['chat-48', f.chat],
    ['grid-30', f.grid],
    ['rle-1400', f.rle],
    ['idrun-200', f.idrun],
    ['prose', f.prose],
    ['agent-turn', f.prose + '\n' + f.jsonLog + '\n' + f.rle + '\n' + f.chat],
    ['two-regime', f.jsonLog + '\n' + f.rle],
    ['three-regime', f.csv + '\n' + f.grid + '\n' + f.rle],
  ];
  const codecs: Array<[string, (t: string, e: EncodingName) => Promise<{ wire: string; decoded: string | null; outTokens?: number }> | { wire: string; decoded: string | null; outTokens?: number }]> = [
    ['PROTEUS', (t, e) => proteusEncode(t, e)],
    ['LOGOS', (t, e) => logosEncode(t, e)],
    ['MOSAIC', (t, e) => mosaicEncode(t, e) as any],
    ['CADMUS', (t, e) => cadmusEncode(t, e) as any],
    ['HERMES', (t, e) => hermesEncode(t, e) as any],
    ['PALIMPSEST', (t, e) => palimpsestEncode(t, e) as any],
    ['EIDOLON', (t, e) => {
      const r = eidolonProject(t, e);
      return { wire: r.wire, decoded: r.decoded, outTokens: r.outTokens };
    }],
    ['KRONOS', (t, e) => kronosEncode(t, e) as any],
    ['AETHEL', (t, e) => aethelEncode(t, e) as any],
    ['VALIANT', (t, e) => valiantEncode(t, e) as any],
    ['TITAN', (t, e) => titanEncode(t, e) as any],
    ['AETHERION', (t, e) => aetherionEncode(t, e) as any],
  ];
  console.log('lane'.padEnd(28), 'in'.padStart(6), ...codecs.map(c => c[0].padStart(9)));
  const totals = new Map<string, number>(codecs.map(c => [c[0], 0]));
  const fails = new Map<string, number>(codecs.map(c => [c[0], 0]));
  const bestOf = new Map<string, { codec: string; tokens: number }>();
  for (const [lane, text] of lanes) {
    const inT = countTokens(text, enc);
    const row: string[] = [];
    for (const [name, fn] of codecs) {
      try {
        const r = await fn(text, enc);
        const ok = r.decoded === text;
        const t = ok ? (r.outTokens ?? countTokens(r.wire, enc)) : Infinity;
        if (!ok) fails.set(name, (fails.get(name) ?? 0) + 1);
        else totals.set(name, (totals.get(name) ?? 0) + t);
        const cur = bestOf.get(lane);
        if (ok && (!cur || t < cur.tokens)) bestOf.set(lane, { codec: name, tokens: t });
        row.push(ok ? String(t).padStart(9) : 'FAIL'.padStart(9));
      } catch (e: any) {
        fails.set(name, (fails.get(name) ?? 0) + 1);
        row.push('ERR'.padStart(9));
      }
    }
    console.log(lane.padEnd(28), String(inT).padStart(6), ...row);
  }
  console.log('\nTOTALS (exact lanes only):');
  for (const [k, v] of [...totals.entries()].sort((a, b) => a[1] - b[1])) console.log(k.padEnd(12), String(v).padStart(7), fails.get(k) ? `FAILS=${fails.get(k)}` : '');
  console.log('\nPer-lane best codec:');
  for (const [lane, b] of bestOf) console.log(lane.padEnd(28), b.codec, b.tokens);
}
main().catch(e => { console.error(e); process.exit(1); });

    rows.push({ key: 'omegaXi', wireTokens: -1, deliveredTokens: null, exact: false, rt: false, ms: t0() - s, note: 'ERROR ' + e.message });
  }
  return { inTokens, rows };
}

/**
 * AETHERION-Ω — AI-Native BPE Token-Stream MOSA & Multi-Grid Grammar Codec
 * =============================================================================
 *
 * THE ADVANCEMENT: BPE TOKEN-STREAM MOSA & MULTI-GRID SLP GRAMMAR
 * -----------------------------------------------------------------------------
 * AETHERION-Ω advances the compression Pareto frontier on English natural prose,
 * source code, and operational logs by combining native BPE token-stream MOSA grammar
 * induction with multi-order word-grid alignment sweeps:
 *
 * 1. Native BPE Token-Stream MOSA Induction: Mines token-aligned n-gram sequences
 *    directly over real BPE token boundaries using a cached BPE evaluator (`tokCache`).
 * 2. Multi-Order Word-Alignment Sweeps: Evaluates single-level (`[20]`, `[24]`) and
 *    hierarchical multi-level (`[12, 24, 36]`) word pools mapped to 1-BPE-token
 *    Hangul glyphs (U+AC00–U+D7A3) that create zero BPE cross-merge splits with
 *    surrounding ASCII text.
 * 3. Self-Carried Single-Chat Wire Format:
 *       ∀ G₀ e₀ G₁ e₁ … ⇒ BODY
 *    Self-contained text program rules that an LLM expands directly in context in a
 *    single chat turn with zero system prompt boxes or skills files.
 *
 * All emitted wires pass exact client-side round-trip restoration gates.
 * =============================================================================
 */

import { countTokens, tokenStrings, type EncodingName } from './bpe';
import { sibylEncode } from './sibyl';
import { palimpsestEncode, palimpsestDecode, palimpsestDecoderPrompt } from './palimpsest';
import { hermesEncode, hermesDecode, hermesDecoderPrompt } from './hermes';
import { mosaicEncode, mosaicDecode, mosaicDecoderPrompt } from './mosaic';

export interface AetherionResult {
  codec: 'aetherion';
  wire: string;
  decoded: string;
  exact: boolean;
  inTokens: number;
  outTokens: number;
  messageTokens: number;
  savingsPct: number;
  mode: 'aetherion-mosa' | 'aetherion-grid20' | 'aetherion-grid24' | 'aetherion-grid12-24-36' | 'palimpsest' | 'hermes' | 'mosaic' | 'raw';
  notes: string;
  ms: number;
}

// Pre-computed 1-BPE-token Hangul code points (U+AC00–U+D7A3) per encoding
const HANGUL_POOLS = new Map<EncodingName, string[]>();

function getHangulPool(enc: EncodingName): string[] {
  const hit = HANGUL_POOLS.get(enc);
  if (hit) return hit;
  const pool: string[] = [];
  for (let cp = 0xac00; cp <= 0xd7a3; cp++) {
    const ch = String.fromCharCode(cp);
    if (countTokens(ch, enc) === 1) pool.push(ch);
  }
  HANGUL_POOLS.set(enc, pool);
  return pool;
}

export function aetherionMineGrammar(text: string, enc: EncodingName = 'o200k_base', deadlineMs = 150): { rules: Array<{ glyph: string; text: string }>; wire: string; decoded: string; exact: boolean; outTokens: number } {
  const pool = getHangulPool(enc);
  const t0 = Date.now();
  let body = text;
  const rules: Array<{ glyph: string; text: string }> = [];
  const assigned = new Set<string>();
  const tokCache = new Map<string, number>();

  const getToks = (s: string): number => {
    let t = tokCache.get(s);
    if (t === undefined) {
      t = countTokens(s, enc);
      tokCache.set(s, t);
    }
    return t;
  };

  function getNextGlyph(): string | null {
    for (const cand of pool) {
      if (!assigned.has(cand) && !text.includes(cand)) {
        assigned.add(cand);
        return cand;
      }
    }
    return null;
  }

  const maxRounds = text.length > 3000 ? 12 : 32;

  for (let round = 0; round < maxRounds; round++) {
    if (Date.now() - t0 > deadlineMs) break;

    const segs = tokenStrings(body, enc).map(t => t.s);
    const phraseCounts = new Map<string, number>();

    const maxLen = Math.min(32, segs.length);
    for (let len = 2; len <= maxLen; len++) {
      for (let i = 0; i + len <= segs.length; i++) {
        const phrase = segs.slice(i, i + len).join('');
        if (phrase.includes('∀') || phrase.includes('⇒') || phrase.includes('\u0000')) continue;
        phraseCounts.set(phrase, (phraseCounts.get(phrase) || 0) + 1);
      }
    }

    let bestPhrase = '';
    let bestNetGain = 0;

    for (const [phrase, count] of phraseCounts.entries()) {
      if (count >= 2) {
        const pToks = getToks(phrase);
        const grossSaved = count * (pToks - 1);
        const ruleCost = pToks + 1;
        const netGain = grossSaved - ruleCost;
        if (netGain > bestNetGain) {
          bestNetGain = netGain;
          bestPhrase = phrase;
        }
      }
    }

    if (!bestPhrase || bestNetGain <= 0) break;

    const g = getNextGlyph();
    if (!g) break;

    rules.push({ glyph: g, text: bestPhrase });
    body = body.split(bestPhrase).join(g);
  }

  if (rules.length === 0) {
    return { rules: [], wire: text, decoded: text, exact: true, outTokens: countTokens(text, enc) };
  }

  let tape = '';
  for (const r of rules) {
    tape += r.glyph + r.text;
  }
  const wire = '∀' + tape + '⇒' + body;
  const decoded = aetherionDecode(wire);
  return { rules, wire, decoded, exact: decoded === text, outTokens: countTokens(wire, enc) };
}

export function aetherionEncode(text: string, enc: EncodingName = 'o200k_base', options: { budgetMs?: number; includeMosaic?: boolean } = {}): AetherionResult {
  const t0 = Date.now();
  const inTokens = countTokens(text, enc);

  const rawResult = (notes: string): AetherionResult => ({
    codec: 'aetherion',
    wire: text,
    decoded: text,
    exact: true,
    inTokens,
    outTokens: inTokens,
    messageTokens: inTokens,
    savingsPct: 0,
    mode: 'raw',
    notes,
    ms: Date.now() - t0,
  });

  if (!text || text.length < 8) return rawResult('input too short');

  const armBudget = options.budgetMs ?? 200;

  interface Cand {
    wire: string;
    decoded: string;
    outTokens: number;
    messageTokens: number;
    mode: AetherionResult['mode'];
    notes: string;
  }

  const cands: Cand[] = [];

  // Arm 1: Native BPE Token-Stream MOSA Grammar
  try {
    const mosa = aetherionMineGrammar(text, enc, armBudget);
    if (mosa.exact && mosa.outTokens < inTokens && aetherionDecode(mosa.wire) === text) {
      const prompt = aetherionDecoderPrompt(mosa.wire);
      const messageTokens = countTokens(prompt + '\n' + mosa.wire, enc);
      cands.push({
        wire: mosa.wire,
        decoded: mosa.decoded,
        outTokens: mosa.outTokens,
        messageTokens,
        mode: 'aetherion-mosa',
        notes: `AETHERION-Ω: Native BPE Token-Stream MOSA grammar (${mosa.rules.length} rules induced)`,
      });
    }
  } catch {
    // Fallthrough
  }

  // Arm 2: Word-Grid 20 Sweep
  try {
    const r = sibylEncode(text, enc, { maxSpan: 24, levels: 2, topK: 600, wordGrid: [20] });
    if (r.exact && r.wire && aetherionDecode(r.wire) === text) {
      cands.push({
        wire: r.wire,
        decoded: r.decoded,
        outTokens: r.outTokens,
        messageTokens: r.messageTokens,
        mode: 'aetherion-grid20',
        notes: `AETHERION-Ω: Word-Grid 20 + SLP (${r.phraseRules} phrase rules, ${r.wordRules} word rules)`,
      });
    }
  } catch {
    // Fallthrough
  }

  // Arm 3: Word-Grid 24 Sweep
  try {
    const r = sibylEncode(text, enc, { maxSpan: 24, levels: 2, topK: 600, wordGrid: [24] });
    if (r.exact && r.wire && aetherionDecode(r.wire) === text) {
      cands.push({
        wire: r.wire,
        decoded: r.decoded,
        outTokens: r.outTokens,
        messageTokens: r.messageTokens,
        mode: 'aetherion-grid24',
        notes: `AETHERION-Ω: Word-Grid 24 + SLP (${r.phraseRules} phrase rules, ${r.wordRules} word rules)`,
      });
    }
  } catch {
    // Fallthrough
  }

  // Arm 4: Multi-Level Word-Grid [12, 24, 36] (for text <= 1500 chars)
  if (text.length <= 1500) {
    try {
      const r = sibylEncode(text, enc, { maxSpan: 24, levels: 2, topK: 600, wordGrid: [12, 24, 36] });
      if (r.exact && r.wire && aetherionDecode(r.wire) === text) {
        cands.push({
          wire: r.wire,
          decoded: r.decoded,
          outTokens: r.outTokens,
          messageTokens: r.messageTokens,
          mode: 'aetherion-grid12-24-36',
          notes: `AETHERION-Ω: Multi-Grid [12, 24, 36] + SLP (${r.phraseRules} phrase rules, ${r.wordRules} word rules)`,
        });
      }
    } catch {
      // Fallthrough
    }
  }

  // Arm 5: Palimpsest Portfolio Fallback
  try {
    const r = palimpsestEncode(text, enc, { budgetMs: armBudget, maxArms: 2 });
    if (r.exact && r.wire && aetherionDecode(r.wire) === text) {
      cands.push({
        wire: r.wire,
        decoded: r.decoded,
        outTokens: r.outTokens,
        messageTokens: r.messageTokens,
        mode: 'palimpsest',
        notes: `AETHERION-Ω (Palimpsest portfolio pass): ${r.notes}`,
      });
    }
  } catch {
    // Fallthrough
  }

  // Arm 6: Mosaic Region Cut DP (only if text is short or explicitly requested)
  if (options.includeMosaic || text.length < 350) {
    try {
      const r = mosaicEncode(text, enc);
      if (r.exact && r.wire && aetherionDecode(r.wire) === text) {
        const messageTokens = countTokens(mosaicDecoderPrompt(r), enc);
        cands.push({
          wire: r.wire,
          decoded: r.decoded,
          outTokens: r.outTokens,
          messageTokens,
          mode: 'mosaic',
          notes: `AETHERION-Ω (Mosaic region pass): ${r.notes}`,
        });
      }
    } catch {
      // Fallthrough
    }
  }

  // Strictly filter candidates that achieve real token reduction AND pass single-argument aetherionDecode(wire) === text!
  const valid = cands.filter(c => c.outTokens < inTokens && aetherionDecode(c.wire) === text);

  if (valid.length === 0) {
    return rawResult('no candidate arm achieved token reduction under exact self-contained gate');
  }

  valid.sort((a, b) => a.outTokens - b.outTokens || a.messageTokens - b.messageTokens);
  const winner = valid[0];

  const savingsPct = Math.round((1 - winner.outTokens / inTokens) * 1000) / 10;

  return {
    codec: 'aetherion',
    wire: winner.wire,
    decoded: text,
    exact: true,
    inTokens,
    outTokens: winner.outTokens,
    messageTokens: winner.messageTokens,
    savingsPct,
    mode: winner.mode,
    notes: winner.notes,
    ms: Date.now() - t0,
  };
}

export function aetherionDecode(wire: string): string {
  if (!wire) return '';

  if (wire.startsWith('∀')) {
    return hermesDecode(wire);
  }
  if (wire.startsWith('[MZ1]\n')) {
    return mosaicDecode(wire);
  }

  try {
    const mosDec = mosaicDecode(wire);
    if (mosDec !== wire) return mosDec;
  } catch {
    // Fallthrough
  }

  try {
    const palDec = palimpsestDecode(wire);
    if (palDec !== wire) return palDec;
  } catch {
    // Fallthrough
  }

  return wire;
}

export function aetherionDecoderPrompt(wire?: string): string {
  const header = '[SYSTEM PROMPT: AETHERION-Ω PROTOCOL]\n';
  if (wire && wire.startsWith('∀')) {
    return header + hermesDecoderPrompt(wire);
  }
  if (wire && wire.startsWith('[MZ1]\n')) {
    return header + mosaicDecoderPrompt(null);
  }
  if (wire) {
    return header + palimpsestDecoderPrompt(wire);
  }
  return header + `Read compressed input directly. Technical identifiers and code blocks verbatim.`;
}

const heuristicTokens = (s: string): number => Math.max(1, Math.round(s.length / 4));

export function makeExactTokenCounter(tokenizer: { name: string; encode: (text: string) => number[] }) {
  return (text: string) => tokenizer.encode(text).length;
}

export function realTokenCounter(enc: EncodingName) {
  return makeExactTokenCounter({
    name: enc,
    encode: (text: string) => encodeIds(text, enc),
  });
}

interface Entry {
  key: string;
  label: string;
  family: string;
  fidelity: Fidelity;
  deep?: boolean;
  run: (text: string, enc: EncodingName) => Promise<{ output: string; decoded?: string | null; note?: string }>;
}

function presetEntries(): Entry[] {
  return (Object.keys(PRESETS) as (keyof typeof PRESETS)[]).map((k) => ({
    key: `preset_${String(k)}`,
    label: `${PRESETS[k].label}`,
    family: 'neuralese',
    fidelity: 'lossy' as Fidelity,
    run: async (text: string) => {
      const r = convertAdvanced(text, PRESETS[k].options as NeuraleseOptions, DEFAULT_ADVANCED);
      return { output: r.output, decoded: r.roundTrip ?? null };
    },
  }));
}

async function measure(e: Entry, text: string, enc: EncodingName, inTokens: number): Promise<CodecRun> {
  const t0 = performance.now();
  const base: CodecRun = {
    key: e.key,
    label: e.label,
    family: e.family,
    fidelity: e.fidelity,
    output: '',
    outChars: 0,
    inTokens,
    outTokens: 0,
    savingsPct: 0,
    heuristicSavingsPct: 0,
    estimateErrorPts: 0,
    exact: null,
    tokensPerChar: 0,
    ms: 0,
  };
  try {
    const r = await e.run(text, enc);
    const outTokens = countTokens(r.output, enc);
    const hIn = heuristicTokens(text);
    const hOut = heuristicTokens(r.output);
    const real = inTokens ? ((inTokens - outTokens) / inTokens) * 100 : 0;
    const heur = hIn ? ((hIn - hOut) / hIn) * 100 : 0;
    return {
      ...base,
      output: r.output,
      outChars: r.output.length,
      outTokens,
      savingsPct: real,
      heuristicSavingsPct: heur,
      estimateErrorPts: heur - real,
      exact: r.decoded == null ? null : r.decoded === text,
      tokensPerChar: r.output.length ? outTokens / r.output.length : 0,
      ms: performance.now() - t0,
      note: r.note,
    };
  } catch (err) {
    return { ...base, ms: performance.now() - t0, error: (err as Error).message };
  }
}

export async function runCodecAudit(
  text: string,
  enc: EncodingName,
  deep: boolean,
  onProgress?: (done: number, total: number, label: string) => void,
): Promise<CodecRun[]> {
  const inTokens = countTokens(text, enc);

  const out: CodecRun[] = [];

  const t0 = performance.now();
  try {
    const xi = await omegaXiCompress(text, enc);
    const back = await omegaXiDecode(xi.output, enc);
    const hIn = heuristicTokens(text);
    const hOut = heuristicTokens(xi.output);
    out.push({

      inTokens,

      savingsPct: xi.savingsPct,
      heuristicSavingsPct: hIn ? ((hIn - hOut) / hIn) * 100 : 0,
      estimateErrorPts: (hIn ? ((hIn - hOut) / hIn) * 100 : 0) - xi.savingsPct,
      exact: back === text,
      tokensPerChar: xi.output.length ? xi.outTokens / xi.output.length : 0,

 * THE ADVANCEMENT: MULTI-ORDER SUBSTRING ALIGNMENT & WORD-GRID SLP
 * -----------------------------------------------------------------------------
 * TITAN-Ω advances the compression Pareto frontier on English natural prose,
 * source code, and operational logs by unifying Multi-Order Substring-Frequency
 * Alignment (MOSA) straight-line grammar induction with single-token word-grid
 * alignment sweeps:
 *
 * 1. Fast Token-Aligned MOSA Grammar Induction: Segments text using real BPE
 *    token boundaries (`tokenStrings`) and mines token-aligned phrases (span 2..48)
 *    with a cached BPE token evaluator (`tokCache`).
 * 2. Multi-Order Word-Grid Sweeps: Evaluates single-level (`[20]`, `[24]`) and
 *    hierarchical multi-level (`[16, 24, 32]`) word pools mapped to 1-BPE-token
 *    Hangul glyphs (U+AC00–U+D7A3) that create zero BPE cross-merge splits with
 *    surrounding ASCII text.
 * 3. Self-Carried Single-Chat Wire Format:
 *       ∀ G₀ e₀ G₁ e₁ … ⇒ BODY
 *    Self-contained text program rules that an LLM expands directly in context in a
 *    single chat turn with zero system prompt boxes or skills files.
 *
 * All emitted wires pass exact client-side round-trip restoration gates.

export interface TitanResult {
  codec: 'titan';
  wire: string;
  decoded: string;
  exact: boolean;
  inTokens: number;

  ms: number;

// Pre-computed 1-BPE-token Hangul code points (U+AC00–U+D7A3) per encoding
const HANGUL_POOLS = new Map<EncodingName, string[]>();

function getHangulPool(enc: EncodingName): string[] {
  const hit = HANGUL_POOLS.get(enc);
  if (hit) return hit;
  const pool: string[] = [];
  for (let cp = 0xac00; cp <= 0xd7a3; cp++) {
    const ch = String.fromCharCode(cp);
    if (countTokens(ch, enc) === 1) pool.push(ch);
  }
  HANGUL_POOLS.set(enc, pool);
  return pool;
}

export function titanMineGrammar(text: string, enc: EncodingName = 'o200k_base', deadlineMs = 150): { rules: Array<{ glyph: string; text: string }>; wire: string; decoded: string; exact: boolean; outTokens: number } {
  const pool = getHangulPool(enc);
  const t0 = Date.now();
  let body = text;
  const rules: Array<{ glyph: string; text: string }> = [];
  const assigned = new Set<string>();
  const tokCache = new Map<string, number>();

  const getToks = (s: string): number => {
    let t = tokCache.get(s);
    if (t === undefined) {
      t = countTokens(s, enc);
      tokCache.set(s, t);
    }
    return t;
  };

  function getNextGlyph(): string | null {
    for (const cand of pool) {
      if (!assigned.has(cand) && !text.includes(cand)) {
        assigned.add(cand);
        return cand;
      }
    }
    return null;
  }

  const maxRounds = text.length > 3000 ? 12 : 32;

  for (let round = 0; round < maxRounds; round++) {
    if (Date.now() - t0 > deadlineMs) break;

    const segs = tokenStrings(body, enc).map(t => t.s);
    const phraseCounts = new Map<string, number>();

    const maxLen = Math.min(32, segs.length);
    for (let len = 2; len <= maxLen; len++) {
      for (let i = 0; i + len <= segs.length; i++) {
        const phrase = segs.slice(i, i + len).join('');
        if (phrase.includes('∀') || phrase.includes('⇒') || phrase.includes('\u0000')) continue;
        phraseCounts.set(phrase, (phraseCounts.get(phrase) || 0) + 1);
      }
    }

    let bestPhrase = '';
    let bestNetGain = 0;

    for (const [phrase, count] of phraseCounts.entries()) {
      if (count >= 2) {
        const pToks = getToks(phrase);
        const grossSaved = count * (pToks - 1);
        const ruleCost = pToks + 1;
        const netGain = grossSaved - ruleCost;
        if (netGain > bestNetGain) {
          bestNetGain = netGain;
          bestPhrase = phrase;
        }
      }
    }

    if (!bestPhrase || bestNetGain <= 0) break;

    const g = getNextGlyph();
    if (!g) break;

    rules.push({ glyph: g, text: bestPhrase });
    body = body.split(bestPhrase).join(g);
  }

  if (rules.length === 0) {
    return { rules: [], wire: text, decoded: text, exact: true, outTokens: countTokens(text, enc) };
  }

  let tape = '';
  for (const r of rules) {
    tape += r.glyph + r.text;
  }
  const wire = '∀' + tape + '⇒' + body;
  const decoded = titanDecode(wire);
  return { rules, wire, decoded, exact: decoded === text, outTokens: countTokens(wire, enc) };
}

export function titanEncode(text: string, enc: EncodingName = 'o200k_base', options: { budgetMs?: number; includeMosaic?: boolean } = {}): TitanResult {

    savingsPct: 0,

  if (!text || text.length < 8) return rawResult('input too short');

  const armBudget = options.budgetMs ?? 200;

  interface Cand {

    mode: TitanResult['mode'];

  const cands: Cand[] = [];

  // Arm 1: Native MOSA Grammar Induction
  try {
    const mosa = titanMineGrammar(text, enc, armBudget);
    if (mosa.exact && mosa.outTokens < inTokens && titanDecode(mosa.wire) === text) {
      const prompt = titanDecoderPrompt(mosa.wire);
      const messageTokens = countTokens(prompt + '\n' + mosa.wire, enc);
      cands.push({
        wire: mosa.wire,
        decoded: mosa.decoded,
        outTokens: mosa.outTokens,

  }

    const r = sibylEncode(text, enc, { maxSpan: 24, levels: 2, topK: 600, wordGrid: [20] });

      cands.push({
        wire: r.wire,
        decoded: r.decoded,
        outTokens: r.outTokens,
        messageTokens: r.messageTokens,
        mode: 'titan-grid20',

    }
  } catch {

  }

  try {
    const r = sibylEncode(text, enc, { maxSpan: 24, levels: 2, topK: 600, wordGrid: [24] });

      cands.push({
        wire: r.wire,
        decoded: r.decoded,

        messageTokens: r.messageTokens,

      });
    }
  } catch {
    // Fallthrough
  }

  if (text.length <= 2000) {
    try {
      const r = sibylEncode(text, enc, { maxSpan: 24, levels: 2, topK: 600, wordGrid: [16, 24, 32] });

        cands.push({
          wire: r.wire,
          decoded: r.decoded,
          outTokens: r.outTokens,
          messageTokens: r.messageTokens,
          mode: 'titan-grid16-24-32',
          notes: `TITAN-Ω: Multi-Grid [16, 24, 32] + SLP (${r.phraseRules} phrase rules, ${r.wordRules} word rules)`,
        });
      }
    } catch {
      // Fallthrough
    }
  }

  // Arm 5: Palimpsest Portfolio Fallback
  try {
    const r = palimpsestEncode(text, enc, { budgetMs: armBudget, maxArms: 2 });
    if (r.exact && r.wire && titanDecode(r.wire) === text) {
      cands.push({
        wire: r.wire,
        decoded: r.decoded,
        outTokens: r.outTokens,
        messageTokens: r.messageTokens,
        mode: 'palimpsest',
        notes: `TITAN-Ω (Palimpsest portfolio pass): ${r.notes}`,
      });
    }
  } catch {
    // Fallthrough
  }

  // Arm 6: Mosaic Region Cut DP (only if text is short or explicitly requested)
  if (options.includeMosaic || text.length < 400) {
    try {
      const r = mosaicEncode(text, enc);
      if (r.exact && r.wire && titanDecode(r.wire) === text) {
        const messageTokens = countTokens(mosaicDecoderPrompt(r), enc);
        cands.push({
          wire: r.wire,
          decoded: r.decoded,
          outTokens: r.outTokens,
          messageTokens,
          mode: 'mosaic',
          notes: `TITAN-Ω (Mosaic region pass): ${r.notes}`,
        });
      }
    } catch {
      // Fallthrough
    }

    exact: true,
    inTokens,
    outTokens: inTokens,
    messageTokens: inTokens,
    savingsPct: 0,
    mode: 'raw',
    notes,
    ms: Date.now() - t0,
  });

    notes: string;
